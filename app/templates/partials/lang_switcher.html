{# The Language switcher, rendered from the site's enabled languages plus the tool's own formatting options. Content, not chrome -- a visitor uses it -- so it wears the site's look and follows the palette. ONE definition, two callers: page.html imports the macro; the config route renders THIS file directly (the {{ ... }} call at the bottom) so a style change swaps in place, exactly like the other marker tools. The dropdown is a native
, so it needs no JavaScript and works with the keyboard. #} {% macro _ls_label(l, opts) -%} {%- if opts.labels == 'flag' %}{{ l.flag_svg | safe }} {%- elif opts.labels == 'code' %}{{ l.code | upper }} {%- elif opts.labels == 'english' %}{{ l.english }} {%- else %}{{ l.native }}{% endif -%} {%- endmacro %} {% macro render_lang_switcher(items, opts, editing=False) %} {%- set opts = opts or {'style':'list','color':'text','flags':'0','labels':'native','align':'left'} -%} {%- if items %} {%- if opts.style == 'dropdown' %} {%- set current = (items | selectattr('current') | list | first) or items[0] %}
{{ _ls_label(current, opts) }}
{%- for l in items %} {{ _ls_label(l, opts) }} {%- endfor %}
{%- else %}
{%- for l in items %} {{ _ls_label(l, opts) }} {%- endfor %}
{%- endif %} {%- elif editing %}
No languages enabled yet — turn some on in Design → Languages and run Translate. This will show them.
{%- endif %} {% endmacro %} {{ render_lang_switcher(items, opts, editing) }}
